 	.cpu arm9tdmi
	.fpu softvfp
	.file	"thread.c"

 	.text
	.align	2, 0
	.global	thread
	.type	thread, %function

thread:
	@ args = 0, pretend = 0, frame = 8
	@ frame_needed = 1, uses_anonymous_args = 0

 	mov	ip, sp
	stmfd	sp!, {fp, ip, lr, pc}
 	sub	fp, ip, #4
	sub	sp, sp, #8

       	stmfd 	sp!, {r2-r11}		/* preserve entry state */
      	ldr   	v3, .L1
      	str   	sp, [v3, #C_SP]

      	mov   	sl, r0			/* setup stack min, DP & RAM top */
	str	r0, [v3, #DP_ADD]
      	add   	r2, r1, r0
	str	r2, [v3, #RAMTOP]

      	b     	origin			/* go initialise TIL */

back2c:
      	ldr   	v3, .L1			/* restore entry state */
      	ldr   	sp, [v3, #C_SP]
      	ldmfd 	sp!, {r2-r11}

 	sub	sp, fp, #12
	ldmfd	sp, {fp, sp, pc}
.L3:

/* ====== End of 'C' interface ======= */


	.include "thread_macros.s"

	.set       BperBUF, (1024 * SCRperBUF)
	.set    BRANCHCODE, 0xEA
 	.set          C_SP, 0x10
	.set        DP_ADD, 0x18
	.set         ENTER, 0x0A	/* 0x0D  !!! */
	.set           ESC, 0x40	/* 0x1B  !!! */
	.set      NBUFFERS, 3
	.set     SCRperBUF, 16
	.set        NWIDTH, 31
	.set        RAMTOP, 0x14
	.set        RP_ADD, 0x08
	.set        RUBOUT, 0x7F
	.set        SP_ADD, 0x04
	.set       TIB_ADD, 0x0C
	.set    TIB_RS_LEN, (1040 + (1024 * 4))
	.set       TOS_GAP, (16 * 4)
	.set       TOR_GAP, (4 * 4)
	.set        UP_ADD, 0x00
	.set       UVARLEN, 0x100


 	.align	2, 0
.L2:	.long	TIB_RS_LEN
.L1:	.long	vm_vars

 	.text
 	.align	2, 0

origin:
	b  	_cold		/* +00 branch to cold-start entry		*/
	b  	_cold		/* +04 branch to warm-start entry		*/
	pfa	b_cold_		/* +08 PFA of cold-start thread			*/
	pfa	abort_		/* +0C PFA "  warm-  "	   "			*/
	addr	x		/* +10 LATEST NFA in FORTH vocab		*/
	.long	RUBOUT		/* +14 Erase key code				*/
				/* 'set_vars_' copies next 4 items from vm_vars	*/
        .long   0		/* +18  Initial user area			*/
        .long   0		/* +1C   "    top of Parameter stack		*/
        .long   0		/* +20   "     "   " Return      "		*/
        .long   0		/* +24   "    Terminal Input Buffer		*/
        .long	NWIDTH      	/* +28  Name field width			*/
        .long   0           	/* +2C  Initial value of WARNING		*/
        addr    x_ + 0x10	/* +30     "      "    " FENCE			*/
        addr    cold_       	/* +34  CFA of COLD word			*/
        addr    _b_emit     	/* +38  CFA of EMIT m/code			*/
        addr    test		/* +3C  LATEST NFA in SYSWORDS vocabulary	*/
        addr    _b_key     	/* +40  CFA of KEY m/code - waits for char i/p	*/


/* ================ Initialise VM ================== */

_cold:
  ldr   v3, .L1
  ldr	r1, [v3, #RAMTOP]
  sub   r1, r1, #UVARLEN
  mov   v4, r1			/* setup VP */
  mov   v5, r1			/* setup UP */
  str   r1, [v3, #UP_ADD]

  sub   r1, r1, #TOR_GAP
  str   r1, [v3, #RP_ADD]
  mov   r9, r1			/* setup RP */

  ldr	r0, .L2
  sub   r1, r1, r0
  str   r1, [v3, #TIB_ADD]	/* setup TIB */

  sub   r1, r1, #TOS_GAP
  str   r1, [v3, #SP_ADD]
  mov   sp, r1			/* setup SP */

  ldr   ip, (origin + 8)	/* setup IP */
  next				/* go start threaded code */

_bfnd1:
  cmp   r0,#0
  beq   _bfndB
  ldrb  r2,[r0]
  and   r2,r2,#63
  ldrb  r5,[r1]
  cmp   r2,r5
  bne   _bfndC
  and   r2,r2,#NWIDTH
  mov   fp,#1
  mov   r5,r2
_bfndF:
  cmp   r5,#0
  beq   _bfndC
  add   r6,r0,r5
  ldrb  r3,[r6]
  and   r3,r3,#127
  add   r6,r1,r5
  ldrb  r4,[r6]
  cmp   r3,r4
  beq   _bfndG
  mov   fp,#0
  mov   r5,#1
_bfndG:
  sub   r5,r5,#1
  b     _bfndF
_bfndC:
  and   r2,r2,#NWIDTH
  add   r0,r0,r2
  mov   r0,r0,LSR#2
  mov   r0,r0,LSL#2
  add   r0,r0,#0x04
  cmp   fp,#0
  beq   _bfndD
  add   r0,r0,#4
  str   r0,[sp]
  mov   r0,#0
  b     _bfndE

  ldr   r0,[r0]
_bfndE:
  b     _bfnd1
_bfndB:
  next

_b_loop:	/* incr .L top of Rstack, true-reltv br-if-count-exceeded */
  mcode
  ldr   r2,[r9]
  add   r2,r2,#1
  str   r2,[r9]
  add   r6,r9,#4
  ldr   r1,[r6]
  cmp   r2,r1
  bge   lp1
  ldr   ip,[ip]
  next
lp1:
  add   r9,r9,#8
  add   ip,ip,#4
  next

  nolink
_b_plus_loop:	/* incr .L top of Rstack by signed val on TOS, true-reltv br-if-count-exceeded */
  mcode
  ldr   fp,[sp],#4
  ldr   r2,[r9]
  add   r2,r2,fp
  str   r2,[r9]
  add   r0,r9,#4
  ldr   r1,[r0]
  mov	  r0,#0
  cmp   r2,r1
  bge   plp1
  mov	  r0,#1
plp1:
  cmp   fp,#0
  bge   plp2
  eor   r0,r0,#1
plp2:  
  cmp   r0,#1
  bne   plp3
  ldr   ip,[ip]
  next
plp3:
  add   r9,r9,#8
  add   ip,ip,#4
  next

_mdiv_mod:
  mcode
/*  pop op		*/
/*  pop r2		*/
  ldr   r1,[sp],#4
  ldr   r2,[sp]
/*  call scheck		*/
/*  set w lsb to 1	*/
  mov   fp,#01
/*  do while op.msb >= 0 */
_bmdv2:
    cmp   r1,#0
    blt  _bmdv1
/*    shift op left 1	*/
  mov   r1,r1,ASL#1
/*    inc w.lsb by 1	*/
  add   fp,fp,#1
/*  enddo		*/
 bal  _bmdv2
_bmdv1:
/*  set ac to 0		*/
  mov   r0,#0
/*  do while w.lsb > 0	*/
 cmp   r0,#0
 /* beq   _bmdv3	*/
/*    shift ac left 1	*/
 mov   r0,r0,ASL#1
/*    sub op from r2	*/
 sub   r2,r2,r1
/*    if borrow = 0	*/
/*       or ac with 1	*/
/*    else		*/
/*       add op to r2	*/
/*    endif		*/
/*    shift op right by 1  */
/*    decrement w.lsb by 1 */
/*  enddo		*/
_bmdv3:
/*  call {sjust}	*/
/*  push r2		*/
  str   r1,[sp]
/*  push ac		*/
  str   r2,[sp,#-4]!
 next
