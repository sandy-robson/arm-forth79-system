file:
  stat   4
  .ascii "fil"
  term   'e'
  link   find
file_:
  thread	/*  read/write -> */
/*
 */
; if 'mode' <> read, report error 0x21
  cfa    _dup
  cfa    _rd_
  cfa    _c_at
  cfa    _minus
  cfa    _b_lit
  val    0x21
  cfa    query_error_
; set fcb mode to 'r'
  cfa    _fcb_mode_
  cfa    _c_store
  cfa    _fcb_name_
  cfa    _fcb_mode_
  cfa    _fopen
  cfa    _dup
; if file open fails, report open error
  cfa    _zero_eq
  cfa    _b_lit
  val    0x20
  cfa    query_error_
; else store file ID in fcb
  cfa    _fcb_fid_
  cfa    _d_store
; get next file buf addr
  cfa    fbuffer_
; store fcb addr in buffer fcb field
  cfa    _fcb_ptr_
  cfa    _d_at
  cfa    _over
  cfa    _two_minus
  cfa    _two_minus
  cfa    _d_store
; store buffer addr in fcb buf field
  cfa    _dup
  cfa    _fcb_buf_
  cfa    _d_store
;  cfa    _dup  ; *** need copy? ***
; set line char index to 0
  cfa    _zero_
; fill 1st buffer 'screen' with bl
;  cfa    _over
;  cfa    _bbuf_
;  cfa    blanks_
; read char from file
fil1:
  cfa    _fcb_fid_
  cfa    _d_at
  cfa    _fgetc
; if not EOF
  cfa    _b_lit
  val    0xFFFFFFFF
  cfa    _over
  cfa    _minus
  cfa    _b_zbranch
  addr   fil4
; if char < bl
  cfa    _dup
  cfa    _bl_
  cfa    _lt
  cfa    _b_zbranch
  addr   fil2
; if char = n/l
  cfa    _n_l_
  cfa    _eq
  cfa    _b_zbranch
  addr   fil3
; fill remaining line chars with bl
  cfa    _over
  cfa    _over
  cfa    _plus
  cfa    _over
  cfa    _cl_
  cfa    _swap
  cfa    _minus
  cfa    blanks_
; discard char-position
  cfa    _drop
; add screen-width to buffer offset
  cfa    _cl_
  cfa    _plus
; reset char_position to 0
  cfa    _zero_
  cfa    _b_branch
  addr   fil3
; {printable char}
fil2:
  cfa    _b_to_r
; ***  incr len by 1  ***
; calc next buffer store position
  cfa    _over
  cfa    _over  
  cfa    _plus
; store char in buffer
  cfa    _b_r_from
  cfa    _swap
  cfa    _c_store
; incr char position in line index by 1
  cfa    _one_plus
; limit to max position
  cfa    _cl_
  cfa    _one_minus
  cfa    _min
; {non-print char, not n/l; *** discard? *** }
fil3:
  cfa    _b_branch
  addr   fil1
; on EOF ...
fil4:
; drop indices
;  cfa    _two_drop  /* only need 3 drops? */
  cfa    _drop
  cfa    _two_drop
; close file
  cfa    _fcb_fid_
  cfa    _d_at
  cfa    _fclose
; set file ID to 0
  cfa    _zero_
  cfa    _fcb_fid_
  cfa    _d_store
  exit
