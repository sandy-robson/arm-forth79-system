            EVEN
main
            mov.b     #0xF9, r0l
            mov.b     r0l, @PCR1

            mov.b     #0x19, r0l
            mov.b     r0l, @PCR2

            mov.b     #0xFD, r0l
            mov.b     r0l, @PCR3

            mov.b     #0xCF, r0l
; !!! SR *** setup PCR5_RAM when using I2C ***
            mov.b     r0l, @PCR5

            mov.b     #0x8F, r0l
            mov.b     r0l, @PCR6

            mov.b     #0xD0, r0l
            mov.b     r0l, @PCR7

            mov.b     #0x00, r0l
            mov.b     r0l, @PCR8

; PDR5 &=~0xC0
            mov.b     @PDR5, r0l
            and.b     #0x3F, r0l
            mov.b     r0l, @PDR5

;            mov.b     #0x04, r0l
;            mov.b     r0l, @PMR1

            mov.b     #0x10, r0l
            mov.b     r0l, @PUCR5
            
; initialize the pseudo_SPI interface
;  PDR3 &= ~0x04 - pre-set CLK to 0
            mov.b     @PDR3, r0l
            and.b     #0xFB, r0l
            mov.b     r0l, @PDR3

;  PDR3 &= ~0x01 - pre-set  DO to 0
            mov.b     @PDR3, r0l
            and.b     #0xFE, r0l
            mov.b     r0l, @PDR3

;  PDR3 |=  BIT3 - pre-set /CS to 1
            mov.b     @PDR3, r0l
            or.b      #0x08, r0l
            mov.b     r0l, @PDR3


;  PMR1 |= 0x01                 //Setup TMOW to output 4MHz clock
            mov.b     @PMR1, r0l
            or.b      #0x01, r0l
            mov.b     r0l, @PMR1

;  RTCCR1 |= 0x80
            mov.b     @RTCCR1, r0l
            or.b      #0x80, r0l
            mov.b     r0l, @RTCCR1

;  TCRV1 = 0x00                 // deselect TRGV operation
            mov.b     #0x00, r0l
            mov.b     r0l, @TCRV1

;  TCSRV = 0x00                 // clear all Tv status bits & deselect TMOV
            mov.b     #0x00, r0l
            mov.b     r0l, @TCNTV
; !!! SR *** can test ****
;  TCRV0 = OVIE | CKS2 | CKS1   // overflow interrupts, falling edge, ext. clock
            mov.b     #0x26, r0l    ; was 0x06
            mov.b     r0l, @TCRV0

;  TCNTV = 0xFF                 // set timer V count to (overflow - 1)
            mov.b     #0xFF, r0l
            mov.b     r0l, @TCNTV

;  TCORA = 0x00                 // set timer V compareA to 0x00
            mov.b     #0x00, r0l
            mov.b     r0l, @TCORA

;  TCORB = 0x00                 // set timer V compareB to 0x00
            mov.b     #0x00, r0l
            mov.b     r0l, @TCORB

; !!! SR *** test csn ****
;  clear can intr. flag
            mov.b     #0x00, r0l
            mov.b     r0l, @can_flag



; ============== Initialise inner interpreter ================

_cold
            mov.b     #0x00,r0l
            mov.b     r0l,@rxin
            mov.b     r0l,@rxout
            mov.b     r0l,@txin
            mov.b     r0l,@txout
            mov.l     #mon_buf,er0
            mov.l     er0,@dp_addr:24       ; set DP to RAM dictionary memory

            mov.l     @r_stack_addr:24,er3  ; setup 'rp'
            mov.l     #_inext:32,er4        ; setup 'inext' reg
            mov.l     @start_thread,er5     ; setup 'ip' reg for cold start
            next

; ================ Interrupts ====================

            RSEG      CODE

; !!! SR *** test can intrs. ***
can_interrupt
            push.l    er0
            
; reset TCNTV to 0xFF
            mov.b     #0xFF,r0l
            mov.b     r0l,@TCNTV
; set can_flag
            mov.b     r0l,@can_flag
; clear TCSRV regr
            mov.b     @TCSRV, r0l
            and.b     #0x00, r0l
            mov.b     r0l, @TCSRV

            pop.l    er0
            rte

; read 2515 err reg
; If either o/flow bit set
;  If RXB0 o/flow bit set And RXB0 full
;   report can RXB0 overrun
;  Endif
;  If RXB1 o/flow bit set And RXB1 full
;   report can RXB1 overrun
;  Endif
;  read & discard i/p msg
;  clear error & int regrs
; Else
;  If RXB0 full
;   If KAN bfr full
;    report KAN i/p buffer full
;    discard CAN RXB0 message
;   Else
;    read RXB0 message into KAN i/p buffer
;   Endif
;  Endif
; Endif
; If RXB1 full
;  If KAN bfr full
;   report KAN i/p buffer full
;   discard CAN RXB1 message
;  Else
;   read RXB1 message into KAN i/p buffer
;  Endif
; Endif

serial_interrupt
            push.l    er0
            push.l    er1
            push.l    er2
       
            mov.b     @SSR,r0l
            and.b     #RDRF,r0l
            cmp       #RDRF,r0l
            bne       stx_int

            mov.b     @RDR,r0l
            mov.l     #rxbuf,er2
            mov.w     #0x0000,e1
            mov.b     #0x00,r1h
            mov.b     @rxin,r1l
; !!! SR *** handle overrun ? ***
            add.l     er1,er2
            mov.b     r0l,@er2
            inc.b     r1l
            mov.b     r1l,@rxin
stx_int
            mov.b     @SCR3,r0l
            and.b     #TX_INT_ENBL,r0l
            cmp       #TX_INT_ENBL,r0l
            bne       exit_int

            mov.b     @SSR,r0l
            and.b     #TDRE,r0l
            cmp       #TDRE,r0l
            bne       exit_int

            mov.b     @txout,r0l
            mov.b     @txin,r0h
            cmp.b     r0l,r0h
            bne       txint_wr

            mov.b     @SCR3,r0h
            and.b     #(0xFF - TX_INT_ENBL),r0h
            mov.b     r0h,@SCR3
            bra       exit_int
txint_wr
            mov.l     #txbuf,er2
            mov.w     #0x0000,e1
            mov.b     #0x00,r1h
            mov.b     @txout,r1l
            add.l     er1,er2
            mov.b     @er2,r0l
            mov.b     r0l,@TDR

            inc.b     r1l
            mov.b     r1l,@txout
            mov.b     @txin,r0l
            cmp.b     r0l,r1l
            bne       exit_int
            
            mov.b     @SCR3,r0h
            and.b     #(0xFF - TX_INT_ENBL),r0h
            mov.b     r0h,@SCR3
exit_int
            pop.l     er2
            pop.l     er1
            pop.l     er0
            rte


default_svc
            rte  ;
