: fcb  ( ${fcbname filename.txt} -> )
  create
    0 d,  0FCB0 , 0 ,  0 d,  0 d,  0 d,
    bl word  1+ dup  1- dup  c@ dup >r cmove
    r> allot  0 c, here align dp d!
  does>
    fcb_ptr d!  ;
: fcb_offs  (n ${fcb_param} -> )
  create
    d,
  does>
    d@ fcb_ptr d@ +  ;
proc file ( mode -> )
  err 21 if mode <> rd
  set fcb_mode to mode
  fopen 'fcb_name 'fcb_mode
  err 20 if fid = 0
  set fcb_fid to fid
  get 1st databyte addr for next free buffer (NFB)
  get addr of current fcb
  set fcb ptr at NFB start to current fcb
  set fcb_buf to 1st databyte addr for NFB
  set fcb_len to 0
  [ addr count -> ]
  do get next file char 
    while rtn <> -1
    if char < bl 
      set char = bl
    endif
    set byte at next position in buffer to char
    increment position by 1
    increment fcb_len by 1
  endwhile
  discard rtn, count, & addr
  fclose fcb_fid
  set fcb_fid to 0
endproc
